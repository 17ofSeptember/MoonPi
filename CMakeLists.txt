cmake_minimum_required(VERSION 3.24)
project(MoonPi VERSION 0.8.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
set(JSON_VALIDATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json_validator GIT_REPOSITORY https://github.com/pboettch/json-schema-validator.git GIT_TAG 2.3.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json_validator)
FetchContent_Declare(civetweb GIT_REPOSITORY https://github.com/civetweb/civetweb.git GIT_TAG v1.16 GIT_SHALLOW TRUE SOURCE_SUBDIR moonpi-no-upstream-build)
FetchContent_MakeAvailable(civetweb)
# Bound incoming frames BEFORE CivetWeb allocates a payload buffer.
file(READ ${civetweb_SOURCE_DIR}/src/civetweb.c CIVET_SOURCE)
set(FRAME_MARKER "/* Allocate space to hold websocket payload */")
string(FIND "${CIVET_SOURCE}" "${FRAME_MARKER}" FRAME_POSITION)
if(FRAME_POSITION EQUAL -1)
  message(FATAL_ERROR "CivetWeb frame-limit patch no longer matches")
endif()
string(REPLACE "${FRAME_MARKER}" "if (data_len > 4096) { break; } ${FRAME_MARKER}" CIVET_SOURCE "${CIVET_SOURCE}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/civetweb-bounded.c "${CIVET_SOURCE}")
add_library(moonpi_http STATIC ${CMAKE_CURRENT_BINARY_DIR}/civetweb-bounded.c)
target_include_directories(moonpi_http PRIVATE ${civetweb_SOURCE_DIR}/src)
target_include_directories(moonpi_http PUBLIC ${civetweb_SOURCE_DIR}/include)
target_compile_definitions(moonpi_http PRIVATE NO_SSL NO_CGI NO_CACHING NO_FILES USE_WEBSOCKET)
find_package(Threads REQUIRED)
target_link_libraries(moonpi_http PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(moonpi_http PUBLIC ws2_32)
endif()
configure_file(cmake/version.hpp.in generated/moonpi/version.hpp @ONLY)
add_library(moonpi_bme280 STATIC third_party/bme280/bme280.c)
target_include_directories(moonpi_bme280 PUBLIC third_party/bme280)
target_compile_definitions(moonpi_bme280 PUBLIC BME280_DOUBLE_ENABLE)
add_library(moonpi_core src/catalog.cpp src/graph.cpp src/hardware.cpp src/gpio_device.cpp src/i2c.cpp src/sensors.cpp src/schedule.cpp src/runtime.cpp src/persistence.cpp src/app.cpp)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS third_party/lua-5.4.9/src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(moonpi_lua STATIC ${LUA_SOURCES})
target_include_directories(moonpi_lua PUBLIC third_party/lua-5.4.9/src)
target_sources(moonpi_core PRIVATE src/security.cpp src/script.cpp src/doctor.cpp)
target_link_libraries(moonpi_core PUBLIC moonpi_lua)
if(WIN32)
  target_link_libraries(moonpi_core PUBLIC bcrypt)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(moonpi_core PRIVATE src/gpio_linux.cpp)
  target_sources(moonpi_core PRIVATE src/i2c_linux.cpp)
endif()
target_include_directories(moonpi_core PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(moonpi_core PUBLIC nlohmann_json::nlohmann_json nlohmann_json_schema_validator Threads::Threads moonpi_bme280)
add_executable(moonpi src/main.cpp src/server.cpp)
target_link_libraries(moonpi PRIVATE moonpi_core moonpi_http)
option(MOONPI_ENABLE_ASAN "Address sanitizer" OFF)
option(MOONPI_ENABLE_UBSAN "Undefined behavior sanitizer (GCC/Clang)" OFF)
foreach(t moonpi_core moonpi)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
    target_compile_definitions(${t} PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MOONPI_ENABLE_ASAN)
      target_compile_options(${t} PRIVATE /fsanitize=address)
    endif()
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
    if(MOONPI_ENABLE_ASAN)
      target_compile_options(${t} PUBLIC -fsanitize=address -fno-omit-frame-pointer)
      target_link_options(${t} PUBLIC -fsanitize=address)
    endif()
    if(MOONPI_ENABLE_UBSAN)
      target_compile_options(${t} PUBLIC -fsanitize=undefined)
      target_link_options(${t} PUBLIC -fsanitize=undefined)
    endif()
  endif()
endforeach()
include(CTest)
option(MOONPI_BUILD_HARDWARE_TESTS "Build opt-in GPIO17 LED test (Linux only)" OFF)
if(BUILD_TESTING)
  add_executable(moonpi_tests tests/core_tests.cpp)
  target_link_libraries(moonpi_tests PRIVATE moonpi_core)
  add_test(NAME core COMMAND moonpi_tests ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(moonpi_gpio_tests tests/gpio_tests.cpp)
  target_link_libraries(moonpi_gpio_tests PRIVATE moonpi_core)
  add_test(NAME gpio_backend COMMAND moonpi_gpio_tests ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(moonpi_advanced_tests tests/advanced_tests.cpp)
  target_link_libraries(moonpi_advanced_tests PRIVATE moonpi_core)
  add_test(NAME advanced_runtime COMMAND moonpi_advanced_tests ${CMAKE_CURRENT_SOURCE_DIR})
  set_tests_properties(core gpio_backend advanced_runtime PROPERTIES TIMEOUT 60)
  if(MOONPI_BUILD_HARDWARE_TESTS AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(moonpi_hardware_test tests/hardware_led.cpp)
    target_link_libraries(moonpi_hardware_test PRIVATE moonpi_core)
    add_test(NAME hardware_led COMMAND moonpi_hardware_test)
    set_tests_properties(hardware_led PROPERTIES LABELS hardware SKIP_RETURN_CODE 77 TIMEOUT 10)
  endif()
endif()
